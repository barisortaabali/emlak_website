ACCESS THE ADMIN PANEL
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Method 1: Click Settings Button
  • Go to your website homepage
  • Look for ⚙️ (settings gear) button in bottom-right corner
  • Click it

Method 2: Direct URL
  • Add "/#/admin" to your website URL
  • Example: https://yourdomain.com/#/admin


LOGIN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Enter your admin email address
2. Enter your admin password
3. Click "Giriş Yap" (Login button)


VIEW APPLICATIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

✓ All property requests appear as cards
✓ Newest applications show first
✓ List updates automatically every 5 seconds
✓ Each card shows:
  - Property type (Satın Alma / Satış)
  - Property details (Konut / Arazi type, size, location)
  - Contact information (name, phone, WhatsApp)
  - Submission date and time
  - Any notes or description


MANAGE APPLICATIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

TO DELETE AN APPLICATION:
  1. Find the application you want to remove
  2. Click the "Sil" (Delete) button at bottom of card
  3. Confirm in the popup: "Bu başvuruyu silmek istediğinize emin misiniz?"
  4. Click "Sil" to delete or "İptal Et" to cancel
  5. Application is permanently removed

COMMON TASKS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

REFRESH APPLICATIONS:
  → Wait 5 seconds (automatic)
  → Or refresh page (F5 / Ctrl+R)

FIND A SPECIFIC APPLICATION:
  → Look for badges with type information
  → Scroll through the list
  → Search by location in your browser (Ctrl+F)

CHECK APPLICATION COUNT:
  → Look at subtitle "X başvuru bulunmaktadır"
  → Shows total number of applications


TROUBLESHOOTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PROBLEM: Can't login
  ✗ Check email spelling
  ✗ Check for CAPS LOCK
  ✗ Try copying/pasting password
  ✗ Try private/incognito browser window
  ✗ Clear browser cache and try again

PROBLEM: Applications not showing
  ✗ Check if new applications were submitted
  ✗ Wait 5 seconds for auto-refresh
  ✗ Refresh page manually (F5)
  ✗ Try logging out and logging back in

PROBLEM: Delete didn't work
  ✗ Check internet connection
  ✗ Try refreshing page
  ✗ Try deleting again
  ✗ Check browser console (F12)

PROBLEM: Page looks broken
  ✗ Refresh page (F5)
  ✗ Clear browser cache
  ✗ Try different browser
  ✗ Close and reopen browser


SECURITY TIPS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

✓ Don't share your login with anyone
✓ ALWAYS logout when finished
✓ Never leave admin panel unattended while logged in
